const renderFAQPreview = (rawContent) => {
  try {
    // Parse the JSON-LD
    const parsed = JSON.parse(
      rawContent
        .replace(/```json\n?/g, '')
        .replace(/```\n?/g, '')
        .replace(/"charCount":\s*\d+/g, '')
        .replace(/,\s*\}(\s*\])/g, '}$1')
        .trim()
    );
    
    const questions = parsed.mainEntity || 
      parsed['@graph']?.[0]?.mainEntity || 
      [];
    
    return (
      <div style={{ 
        maxHeight: 300, 
        overflowY: 'auto',
        padding: '12px 0'
      }}>
        {questions.map((q, i) => (
          <div key={i} style={{ 
            marginBottom: 16,
            paddingBottom: 16,
            borderBottom: '1px solid #f0f0f0'
          }}>
            <p style={{ 
              fontWeight: 500,
              fontSize: 13,
              color: '#1a1a2e',
              marginBottom: 4
            }}>
              Q: {q.name}
            </p>
            <p style={{ 
              fontSize: 13,
              color: '#6b7280',
              lineHeight: 1.5
            }}>
              A: {q.acceptedAnswer?.text
                ?.slice(0, 200)}
              {q.acceptedAnswer?.text?.length > 200 
                ? '...' : ''}
            </p>
          </div>
        ))}
      </div>
    );
  } catch (e) {
    // Fallback: show cleaned text
    return (
      <pre style={{ 
        fontSize: 12,
        whiteSpace: 'pre-wrap',
        color: '#374151'
      }}>
        {rawContent
          .replace(/"charCount":\s*\d+/g,'')
          .replace(/\\"/g, '"')
          .replace(/\\n/g, '\n')
          .slice(0, 600)}...
      </pre>
    );
  }
};

For Org Schema:

const renderOrgPreview = (rawContent) => {
  try {
    const parsed = JSON.parse(rawContent
      .replace(/```json\n?/g,'')
      .replace(/```\n?/g,'')
      .trim()
    );
    
    const fields = [
      { label: 'Name', value: parsed.name },
      { label: 'URL', value: parsed.url },
      { label: 'Type', value: parsed['@type'] },
      { label: 'Description', 
        value: parsed.description?.slice(0,150) },
      { label: 'Founded', value: parsed.foundingDate },
    ].filter(f => f.value);
    
    return (
      <div style={{ padding: '8px 0' }}>
        {fields.map(f => (
          <div key={f.label} style={{
            display: 'flex',
            gap: 12,
            padding: '6px 0',
            borderBottom: '1px solid #f0f0f0',
            fontSize: 13
          }}>
            <span style={{ 
              color: '#9ca3af', 
              width: 100,
              flexShrink: 0
            }}>
              {f.label}
            </span>
            <span style={{ color: '#1a1a2e' }}>
              {f.value}
            </span>
          </div>
        ))}
      </div>
    );
  } catch {
    return <pre>{rawContent.slice(0,400)}</pre>;
  }
};
