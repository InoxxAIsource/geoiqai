BACKEND: /api/ai-pr/get-contact
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

New endpoint called when user clicks 
journalist card or opens profile modal.

app.post('/api/ai-pr/get-contact', 
async (req, res) => {
  try {
    const { firstName, lastName, domain } = req.body;
    
    // Check Supabase cache first
    const { data: cached } = await supabase
      .from('journalist_contacts')
      .select('*')
      .eq('name', `${firstName} ${lastName}`)
      .eq('domain', domain)
      .single();
    
    if (cached) {
      return res.json({ 
        success: true, 
        contact: cached,
        from_cache: true 
      });
    }
    
    const results = {};
    
    // Step 1: Hunter Email Finder
    // Find email from name + domain
    try {
      const hunterUrl = new URL(
        'https://api.hunter.io/v2/email-finder'
      );
      hunterUrl.searchParams.set('domain', domain);
      hunterUrl.searchParams.set('first_name', firstName);
      hunterUrl.searchParams.set('last_name', lastName);
      hunterUrl.searchParams.set(
        'api_key', 
        process.env.HUNTER_API_KEY
      );
      
      const hunterResp = await fetch(hunterUrl.toString());
      const hunterData = await hunterResp.json();
      
      if (hunterData.data?.email) {
        results.email = hunterData.data.email;
        results.email_confidence = 
          hunterData.data.score; // 0-100
        results.email_type = 
          hunterData.data.type; // personal or generic
        results.email_verified = 
          hunterData.data.verification?.status === 'valid';
      }
    } catch (e) {
      console.log('Hunter email lookup failed:', e.message);
    }
    
    // Step 2: Hunter Domain Search
    // Get all emails at this outlet domain
    // Use to find journalist even if name lookup fails
    if (!results.email) {
      try {
        const domainUrl = new URL(
          'https://api.hunter.io/v2/domain-search'
        );
        domainUrl.searchParams.set('domain', domain);
        domainUrl.searchParams.set('limit', '10');
        domainUrl.searchParams.set(
          'api_key',
          process.env.HUNTER_API_KEY
        );
        
        const domainResp = await fetch(domainUrl.toString());
        const domainData = await domainResp.json();
        
        // Find matching journalist in results
        const match = domainData.data?.emails?.find(e => 
          e.first_name?.toLowerCase() === 
            firstName.toLowerCase() ||
          e.last_name?.toLowerCase() === 
            lastName.toLowerCase()
        );
        
        if (match) {
          results.email = match.value;
          results.email_confidence = match.confidence;
          results.email_type = match.type;
        }
        
        // Also get domain pattern for manual outreach
        results.email_pattern = 
          domainData.data?.pattern;
        // e.g. "{first}.{last}@searchengineland.com"
        
      } catch (e) {
        console.log('Hunter domain search failed:', e.message);
      }
    }
    
    // Step 3: Find Twitter/LinkedIn via Exa
    try {
      const exaResp = await exa.search(
        `${firstName} ${lastName} journalist 
         ${domain} Twitter LinkedIn contact`,
        {
          type: 'auto',
          numResults: 3,
          category: 'people',
          contents: {
            highlights: {
              numSentences: 2
            }
          }
        }
      );
      
      // Extract Twitter handle from results
      const allText = exaResp.results
        .map(r => [
          r.url, 
          r.title,
          ...(r.highlights || [])
        ].join(' '))
        .join(' ');
      
      // Find Twitter handle
      const twitterMatch = allText.match(
        /twitter\.com\/([a-zA-Z0-9_]+)/
      ) || allText.match(/@([a-zA-Z0-9_]+)/);
      
      if (twitterMatch) {
        results.twitter = `@${twitterMatch[1]}`;
        results.twitter_url = 
          `https://twitter.com/${twitterMatch[1]}`;
      }
      
      // Find LinkedIn URL
      const linkedinResult = exaResp.results
        .find(r => r.url?.includes('linkedin.com/in/'));
      
      if (linkedinResult) {
        results.linkedin_url = linkedinResult.url;
        // Extract handle from URL
        const handle = linkedinResult.url
          .split('linkedin.com/in/')[1]
          ?.replace(/\/$/, '');
        results.linkedin_handle = handle;
      }
      
    } catch (e) {
      console.log('Exa people search failed:', e.message);
    }
    
    // Step 4: Build email from pattern if no email found
    if (!results.email && results.email_pattern) {
      const pattern = results.email_pattern;
      const guessed = pattern
        .replace('{first}', firstName.toLowerCase())
        .replace('{last}', lastName.toLowerCase())
        .replace('{f}', firstName[0].toLowerCase())
        .replace('{l}', lastName[0].toLowerCase());
      
      results.email_guessed = guessed;
      results.email_confidence = 40;
      results.email_note = 
        'Generated from outlet email pattern';
    }
    
    // Cache result in Supabase
    const contactData = {
      name: `${firstName} ${lastName}`,
      first_name: firstName,
      last_name: lastName,
      domain,
      email: results.email || results.email_guessed,
      email_confidence: results.email_confidence,
      email_type: results.email_type,
      email_verified: results.email_verified,
      email_note: results.email_note,
      twitter: results.twitter,
      twitter_url: results.twitter_url,
      linkedin_url: results.linkedin_url,
      looked_up_at: new Date().toISOString()
    };
    
    await supabase
      .from('journalist_contacts')
      .upsert(contactData);
    
    res.json({ 
      success: true, 
      contact: contactData,
      from_cache: false
    });
    
  } catch (error) {
    console.error('Contact lookup error:', error);
    res.status(500).json({ error: error.message });
  }
});

Supabase table:
journalist_contacts (
  id, name, first_name, last_name, domain,
  email, email_confidence, email_type,
  email_verified, email_note,
  twitter, twitter_url, linkedin_url,
  looked_up_at
)
