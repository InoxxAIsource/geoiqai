const checkBrandInAIO = (aioData, domain) => {
  if (!aioData) return {
    cited: false,
    mentioned: false
  };
  
  const cleanDomain = domain
    .replace('www.','').toLowerCase();
  const brandName = cleanDomain.split('.')[0];
  
  // Check in text
  const text = (
    aioData.text || 
    aioData.description || 
    aioData.snippet || ''
  ).toLowerCase();
  
  // Check in links/references
  const links = aioData.links || 
    aioData.references || 
    aioData.items || [];
  
  const citedInLinks = links.some(
    (l) => 
      l.url?.includes(cleanDomain) ||
      l.domain?.includes(cleanDomain)
  );
  
  const mentionedInText = 
    text.includes(cleanDomain) ||
    text.includes(brandName);
  
  return {
    cited: citedInLinks,
    mentioned: mentionedInText || citedInLinks,
    text: aioData.text?.slice(0, 300),
    query: 'Coinbase review'
  };
};
