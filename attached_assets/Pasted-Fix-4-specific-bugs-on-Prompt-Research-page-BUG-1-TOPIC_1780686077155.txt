const topicMap = {}
brandRelevantItems.forEach(item => {
  const query = item.se_query || item.question || ''
  
  // Extract topic from query by removing brand name
  // "netflix streaming services" → "Streaming Services"
  // "how to cancel netflix" → "Cancel Netflix"
  
  // Use first meaningful phrase as topic
  const cleanQuery = query
    .toLowerCase()
    .replace(/^(how to|what is|what are|how do|where is|when does|why is|best|top|free)/i, '')
    .trim()
  
  // Group by first 2-3 meaningful words
  const words = cleanQuery.split(' ').filter(w => w.length > 2)
  const topicKey = words.slice(0, 3).join(' ')
  
  if (!topicMap[topicKey]) {
    // Capitalize properly for display
    const topicDisplay = words.slice(0, 4)
      .map(w => w.charAt(0).toUpperCase() + w.slice(1))
      .join(' ')
    
    topicMap[topicKey] = {
      topic: topicDisplay,
      prompts: [],
      totalAiVolume: 0,
      brands: new Set(),
      sources: new Set()
    }
  }
  
  topicMap[topicKey].prompts.push(item)
  topicMap[topicKey].totalAiVolume += item.ai_search_volume || 0
})

// Fallback: extract capitalized words from answer
// as potential brand names
function extractBrandsFromText(text) {
  if (!text) return []
  // Match capitalized words that appear multiple times
  const words = text.match(/\b[A-Z][a-z]+(?:\s[A-Z][a-z]+)?\b/g) || []
  const freq = {}
  words.forEach(w => freq[w] = (freq[w]||0) + 1)
  return Object.entries(freq)
    .filter(([,count]) => count > 1)
    .sort((a,b) => b[1]-a[1])
    .map(([name, count]) => ({ name, count }))
    .slice(0, 20)
}

const allBrandMap = {}
allItems.forEach(item => {
  // Try brand_entities first
  const entities = item.brand_entities || []
  if (entities.length > 0) {
    entities.forEach(b => {
      if (b.name) {
        allBrandMap[b.name] = (allBrandMap[b.name]||0) + 1
      }
    })
  } else {
    // Fallback to text extraction
    const extracted = extractBrandsFromText(item.answer)
    extracted.forEach(b => {
      allBrandMap[b.name] = (allBrandMap[b.name]||0) + b.count
    })
  }
})

function getRelevance(topicName, brandName) {
  const topic = topicName.toLowerCase()
  const brand = brandName.toLowerCase()
  
  if (topic.includes(brand)) return 'High'
  
  const brandWords = brand.split(/[\s-]/)
  if (brandWords.some(w => w.length > 3 && topic.includes(w))) 
    return 'High'
  
  // Medium for loosely related

}
