function showPage(id, navEl) {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
  const pg = document.getElementById('page-' + id);
  if (pg) pg.classList.add('active');
  if (navEl) navEl.classList.add('active');
  else {
    const n = document.querySelector('[data-page="' + id + '"]');
    if (n) n.classList.add('active');
  }
  initPageCharts(id);
}

// ═══════════════════════════════════════════════
// DOMAIN SEARCH
// ═══════════════════════════════════════════════
function runSearch() {
  const d = document.getElementById('domainInput').value.trim();
  if (!d) return;
  state.domain = d;
  document.getElementById('vis-domain').textContent = d;
  document.getElementById('audit-domain').textContent = d;
  document.getElementById('track-domain').textContent = d;
  document.getElementById('content-domain').textContent = d;
  loadVisibility();
}

function geoChip(geo, el) {
  document.querySelectorAll('.chips .chip').forEach((c,i) => { if(i<4) c.classList.remove('active'); });
  el.classList.add('active');
  state.geo = geo;
  loadVisibility();
}

// ═══════════════════════════════════════════════
// VISIBILITY
// ═══════════════════════════════════════════════
async function loadVisibility() {
  document.getElementById('topics-table-wrap').innerHTML = '<div class="spinbox"><div class="spin"></div><div class="stxt">Loading…</div></div>';
  try {
    const res = await fetch(`/api/visibility?domain=${encodeURIComponent(state.domain)}&geo=${state.geo}&period=${state.period}`);
    const data = await res.json();
    if (data.error) throw new Error(data.error);
    renderVisibility(data);
    document.getElementById('vis-banner').style.display = 'none';
  } catch(e) {
    document.getElementById('vis-banner').style.display = 'flex';
    renderVisibility(getDemoVisibility());
  }
}

function renderVisibility(data) {
  document.getElementById('vis-score').textContent = data.score ?? '—';
  document.getElementById('k-mentions').textContent = data.mentions ?? 0;
  document.getElementById('k-citations').textContent = data.citations ?? 0;
  document.getElementById('k-cited').textContent = data.citedPages ?? 0;
  document.getElementById('k-mentions-ch').textContent = data.mentionsChange ?? '';
  document.getElementById('k-citations-ch').textContent = data.citationsChange ?? '';
  document.getElementById('k-cited-ch').textContent = data.citedPagesChange ?? '';
  document.getElementById('cnt-p').textContent = data.topics?.length ?? 0;
  document.getElementById('cnt-cp').textContent = data.citedPages ?? 0;
  renderLLMTable(data.llm ?? getDemoLLM(), state.llmTab);
  renderTopicsTable(data.topics ?? getDemoTopics());
  updateVisChart(data.trend ?? getDemoTrend());
}

function renderLLMTable(llm, tab) {
  const rows = llm.map(l => {
    const val = tab === 'mentions' ? l.mentionsPct : l.citedPct;
    return `<div class="lrow">
      <div class="lname">${l.icon} ${l.name}</div>
      <div class="lbar"><div class="lfill" style="width:${val}%"></div></div>
      <div class="lval">${val}%</div>
    </div>`;
  }).join('');
  document.getElementById('llm-table').innerHTML = rows;
}

function switchLLM(tab, el) {
  document.querySelectorAll('#page-visibility .tbar:nth-of-type(2) .tbtn').forEach(b => b.classList.remove('active'));
  el.classList.add('active');
  state.llmTab = tab;
  renderLLMTable(getDemoLLM(), tab);
}

function switchTP(p, el) {
  document.querySelectorAll('#page-visibility .tbar:first-of-type .tbtn').forEach(b => b.classList.remove('active'));
  el.classList.add('active');
  state.period = p;
  loadVisibility();
}

function switchTopTab(tab, el) {
  document.querySelectorAll('#page-visibility .tbar:nth-of-type(3) .tbtn').forEach(b => b.classList.remove('active'));
  el.classList.add('active');
  state.topTab = tab;
  if (tab === 'cited-pages') renderCitedPages(getDemoCitedPages());
  else renderTopicsTable(getDemoTopics());
}

async function runCompetitor() {
  const main = document.getElementById('comp-main').value.trim() || state.domain;
  const d1 = document.getElementById('comp-d1').value.trim();
  const d2 = document.getElementById('comp-d2').value.trim();
  const d3 = document.getElementById('comp-d3').value.trim();
  const domains = [main, d1, d2, d3].filter(Boolean);
  if (domains.length < 2) { alert('Enter at least one competitor domain'); return; }

  document.getElementById('comp-empty').style.display = 'none';
  document.getElementById('comp-results').style.display = 'block';

  try {
    const res = await fetch('/api/competitor', { method: 'POST', headers: {'Content-Type':'application/json'}, body: JSON.stringify({ domains }) });
    const data = await res.json();
    if (data.error) throw new Error(data.error);
    renderCompetitor(data, domains);
  } catch(e) {
    renderCompetitor(getDemoCompetitor(domains), domains);
  }
}

function renderCompetitor(data, domains) {
  const scoresWrap = document.getElementById('comp-scores-wrap');
  scoresWrap.innerHTML = domains.map((d, i) => `
    <div class="cscore">
      <div class="cscn">${i === 0 ? 'YOU' : 'COMPETITOR ' + i}</div>
      <div class="cscv ${i===0?'you':''}">${data.scores[i] ?? 0}</div>
      <div style="font-size:11px;color:var(--muted)">${d}</div>
    </div>`).join('');

  if (state.charts.comp) { state.charts.comp.destroy(); delete state.charts.comp; }
  const ctx = document.getElementById('compChart');
  if (ctx) {
    const colors = ['#5B4FDB','#EF4444','#F59E0B','#10B981'];
    state.charts.comp = new Chart(ctx, {
      type: 'line',
      data: {
        labels: data.trend.labels,
        datasets: domains.map((d, i) => ({
          label: d, data: data.trend.series[i] ?? [],
          borderColor: colors[i], backgroundColor: 'transparent', tension: .4, pointRadius: 3
        }))
      },
      options: { responsive: true, plugins: { legend: { position: 'bottom', labels: { font: { family: 'Sora', size: 11 } } } } }
    });
  }

  document.getElementById('comp-insights').innerHTML = (data.insights || []).map((ins, i) => `
    <div class="ii"><div class="inum">${i+1}</div><div class="ibody"><div class="ititle">${ins.title}</div><div class="idesc">${ins.desc}</div></div></div>`).join('');

  document.getElementById('comp-gap-body').innerHTML = (data.gaps || []).map(g => `
    <tr>
      <td>${g.topic}</td>
      <td>${g.aiVolume}</td>
      <td><span class="badge ${g.status==='Missing'?'br':g.status==='Cited'?'bg':'bo'}">${g.status}</span></td>
      <td style="font-size:12px;color:var(--muted)">${g.competitors}</td>
      <td style="font-weight:600;color:var(--red)">${g.missing?.toLocaleString()}</td>
    </tr>`).join('');
}

function quickPrompt(t) { document.getElementById('prompt-input').value = t; runPrompt(); }

async function runPrompt() {
  const topic = document.getElementById('prompt-input').value.trim();
  if (!topic) return;
  document.getElementById('prompt-landing').style.display = 'none';
  document.getElementById('prompt-results').style.display = 'flex';
  document.getElementById('prompt-topic-title').textContent = topic;
  document.getElementById('prompt-kpis').innerHTML = '<div class="spinbox" style="grid-column:span 6"><div class="spin"></div></div>';

  try {
    const res = await fetch(`/api/prompts?topic=${encodeURIComponent(topic)}&geo=${state.geo}`);
    const data = await res.json();
    if (data.error) throw new Error(data.error);
    renderPromptResults(data);
  } catch(e) {
    renderPromptResults(getDemoPrompt(topic));
  }
}
