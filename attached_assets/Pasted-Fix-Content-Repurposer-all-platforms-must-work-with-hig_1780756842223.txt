BACKEND ROUTE IMPLEMENTATION:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

app.post('/api/repurpose', async (req, res) => {
  try {
    const { content, platform } = req.body;
    
    console.log('Repurpose request:', { platform, 
      contentLength: content?.length });
    
    if (!content || !platform) {
      return res.status(400).json({ 
        error: 'Missing content or platform' 
      });
    }
    
    const promptTemplate = platformPrompts[platform];
    if (!promptTemplate) {
      return res.status(400).json({ 
        error: `Unknown platform: ${platform}` 
      });
    }
    
    const userPrompt = promptTemplate.replace(
      '{{content}}', 
      content.slice(0, 3000)
    );
    
    const response = await anthropic.messages.create({
      model: 'claude-opus-4-5',
      max_tokens: 2000,
      system: `You are an expert content strategist 
who writes platform-native, AI-optimized content.
Return ONLY raw JSON. No markdown. No backticks. 
No explanation. No preamble.`,
      messages: [{ role: 'user', content: userPrompt }]
    });
    
    const rawText = response.content[0].text;
    console.log('Claude raw response:', rawText);
    
    // Safe parser
    const cleaned = rawText
      .replace(/```json\n?/g, '')
      .replace(/```\n?/g, '')
      .trim();
    
    const parsed = JSON.parse(cleaned);
    res.json({ success: true, data: parsed });
    
  } catch (error) {
    console.error('Repurpose error:', error.message);
    console.error('Stack:', error.stack);
    res.status(500).json({ error: error.message });
  }
});
