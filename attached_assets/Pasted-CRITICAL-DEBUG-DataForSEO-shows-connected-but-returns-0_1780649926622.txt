New score calculation using TRACK B:

function calculateScore(llmResults) {
  let score = 0;
  const maxPerPrompt = 20; // 5 prompts × 20 = 100
  
  llmResults.forEach(result => {
    let promptScore = 0;
    if (result.mentioned) promptScore += 12;
    if (result.cited) promptScore += 5;
    if (result.sentiment === 'positive') promptScore += 3;
    score += Math.min(promptScore, maxPerPrompt);
  });
  
  return Math.min(Math.round(score), 100);
}
