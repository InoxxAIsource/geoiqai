// GET /api/visibility?domain=example.com&geo=worldwide&period=1m
// ═══════════════════════════════════════════════════════════════
app.get('/api/visibility', async (req, res) => {
  const { domain, geo = 'worldwide', period = '1m' } = req.query;
  if (!domain) return res.status(400).json({ error: 'domain is required' });

  try {
    const api = new DataForSEO();

    // 1. Get aggregated metrics (Citations, Mentions, Cited Pages)
    const metricsRes = await api.post('/v3/ai_optimization/llm_mentions/aggregated_metrics/live', [{
      target: domain,
      date_from: getPeriodStart(period),
      date_to: getToday(),
      location_code: geoToLocation(geo),
      language_code: 'en'
    }]);

    // 2. Get LLM distribution
    const llmRes = await api.post('/v3/ai_optimization/llm_mentions/cross_aggregated_metrics/live', [{
      target: domain,
      date_from: getPeriodStart(period),
      date_to: getToday(),
      group_by: 'llm_name',
      location_code: geoToLocation(geo),
      language_code: 'en'
    }]);

    // 3. Get topics/sources
    const topicsRes = await api.post('/v3/ai_optimization/llm_mentions/search/live', [{
      target: domain,
      date_from: getPeriodStart(period),
      date_to: getToday(),
      location_code: geoToLocation(geo),
      language_code: 'en',
      limit: 50
    }]);

    // 4. Get cited pages
    const citedRes = await api.post('/v3/ai_optimization/llm_mentions/top_pages/live', [{
      target: domain,
      date_from: getPeriodStart(period),
      date_to: getToday(),
      location_code: geoToLocation(geo),
      language_code: 'en',
      limit: 20
    }]);

    // Parse metrics
    const metrics = metricsRes?.tasks?.[0]?.result?.[0] || {};
    const llmData = llmRes?.tasks?.[0]?.result || [];
    const topicsData = topicsRes?.tasks?.[0]?.result || [];
    const citedData = citedRes?.tasks?.[0]?.result || [];

    // Parse LLM distribution
    const llmNames = {
      'chatgpt': { name: 'ChatGPT', icon: '🤖' },
      'gemini': { name: 'Gemini', icon: '💎' },
      'perplexity': { name: 'Perplexity', icon: '🔍' },
      'claude': { name: 'Claude', icon: '✨' },
      'copilot': { name: 'Copilot', icon: '🪟' }
    };

    const totalCitations = metrics.citations_count || 0;
    const llm = llmData.map(item => {
      const key = (item.llm_name || '').toLowerCase();
      const info = llmNames[key] || { name: item.llm_name, icon: '🤖' };
      const pct = totalCitations > 0 ? Math.round((item.citations_count / totalCitations) * 100) : 0;
      return {
        name: info.name,
        icon: info.icon,
        mentionsPct: pct,
        citedPct: Math.round((item.cited_pages_count || 0) / Math.max(metrics.cited_pages_count || 1, 1) * 100)
      };
    });

    // Parse topics
    const topics = topicsData.slice(0, 20).map(t => ({
      topic: t.keyword || t.topic || 'Unknown',
      flag: '🌐',
      visibility: Math.round((t.citations_count || 0) / Math.max(totalCitations, 1) * 100),
      mentions: t.mentions_count || 0,
      aiVolume: formatNumber(t.search_volume || 0),
      intent: t.intent || 'Informational',
      samplePrompt: t.sample_prompt || '—',
      aiResponse: t.ai_response_preview || '—',
      brands: t.brands_count || 0,
      sources: t.sources_count || 0
    }));

    // Parse cited pages
    const citedPages = citedData.slice(0, 16).map(p => ({
      url: p.page_url || p.url || domain,
      prompts: p.prompts_count || 0,
      aiVolume: formatNumber(p.search_volume || 0)
    }));

    // Build trend data (last 6 data points)
    const trend = buildTrend(metrics, period);

    res.json({
      domain,
      score: calculateScore(metrics),
      mentions: metrics.mentions_count || 0,
      citations: totalCitations,
      citedPages: metrics.cited_pages_count || 0,

      topics: topics.length > 0 ? topics : [],
      citedPagesList: citedPages,
      trend
    });

  } catch (err) {
    console.error('Visibility API error:', err.message);
    res.status(500).json({ error: err.message, demo: true });
  }

app.post('/api/competitor', async (req, res) => {
  const { domains, geo = 'worldwide', period = '6m' } = req.body;
  if (!domains || domains.length < 2) return res.status(400).json({ error: 'At least 2 domains required' });

  try {
    const api = new DataForSEO();

    // Get metrics for each domain
    const tasks = domains.map(domain => ({
      target: domain,
      date_from: getPeriodStart(period),
      date_to: getToday(),
      location_code: geoToLocation(geo),
      language_code: 'en'
    }));

    const results = await Promise.all(
      tasks.map(task => api.post('/v3/ai_optimization/llm_mentions/aggregated_metrics/live', [task]))
    );

    const scores = results.map(r => {
      const m = r?.tasks?.[0]?.result?.[0] || {};
      return calculateScore(m);
    });

    // Get monthly trend for all domains
    const trendRes = await api.post('/v3/ai_optimization/llm_mentions/cross_aggregated_metrics/live', [{
      targets: domains,
      date_from: getPeriodStart('6m'),
      date_to: getToday(),
      group_by: 'date',
      location_code: geoToLocation(geo),
      language_code: 'en'
    }]);

    const trendData = trendRes?.tasks?.[0]?.result || [];
    const labels = [...new Set(trendData.map(t => t.date?.substring(0, 7) || ''))].slice(-6);
    const series = domains.map(domain => {
      return labels.map(label => {
        const found = trendData.find(t => t.target === domain && t.date?.startsWith(label));
        return found?.citations_count || 0;
      });
    });

    // Gap analysis
    const gapRes = await api.post('/v3/ai_optimization/llm_mentions/search/live', [{
      targets: domains,
      date_from: getPeriodStart('3m'),
      date_to: getToday(),
      location_code: geoToLocation(geo),
      language_code: 'en',
      limit: 20
    }]);

    const gapData = gapRes?.tasks?.[0]?.result || [];
    const gaps = gapData.map(g => ({
      topic: g.keyword || g.topic || 'Unknown',
      aiVolume: formatNumber(g.search_volume || 0),
      status: g.your_brand_cited ? 'Cited' : g.weak_presence ? 'Weak' : 'Missing',
      competitors: (g.competitor_domains || []).join(', ') || domains.slice(1).join(', '),
      missing: g.missing_prompts_count || 0
    }));

    const insights = buildCompInsights(domains, scores);

    res.json({
      domains,
      scores,
      trend: { labels, series },
      insights,
      gaps: gaps.length > 0 ? gaps : getDemoGaps(domains)
    });

  } catch (err) {
    console.error('Competitor API error:', err.message);
    res.status(500).json({ error: err.message, demo: true });
  }
});

  if (!topic) return res.status(400).json({ error: 'topic is required' });

  try {
    const api = new DataForSEO();

    // AI keyword search volume
    const volRes = await api.post('/v3/ai_optimization/ai_keyword_data/keywords_search_volume/live', [{
      keywords: [topic],
      location_code: geoToLocation(geo),
      language_code: 'en'
    }]);

    // Related topics
    const relRes = await api.post('/v3/ai_optimization/ai_keyword_data/keywords_for_keywords/live', [{
      keywords: [topic],
      location_code: geoToLocation(geo),
      language_code: 'en',
      limit: 20
    }]);

    const volData = volRes?.tasks?.[0]?.result?.[0] || {};
    const relData = relRes?.tasks?.[0]?.result || [];

    const intentMap = {};
    relData.forEach(k => {
      const intent = k.intent || 'Informational';
      intentMap[intent] = (intentMap[intent] || 0) + 1;
    });
    const total = Object.values(intentMap).reduce((a, b) => a + b, 0) || 1;
    const intentPcts = {};
    Object.keys(intentMap).forEach(k => intentPcts[k] = Math.round(intentMap[k] / total * 100));

    const topicsList = relData.map(k => ({
      topic: k.keyword || '—',
      aiVolume: formatNumber(k.search_volume || 0),
      difficulty: (k.difficulty || 50) + '%',
      intent: k.intent || 'Informational',
      brands: (k.top_brands || []).join(', ') || '—',
      trend: k.trend_direction || 'stable'
    }));

    const brands = [...new Set(relData.flatMap(k => k.top_brands || []))].slice(0, 10);
    const sources = [...new Set(relData.flatMap(k => k.top_sources || []))].slice(0, 10);

    res.json({
      topic,
      aiVolume: formatNumber(volData.search_volume || 0),
      topicsCount: formatNumber(relData.length * 50),
      promptsCount: formatNumber(relData.length * 2000),
      brandsCount: formatNumber(brands.length * 1000),
      sourcesCount: formatNumber(sources.length * 1000),
      avgDifficulty: Math.round(relData.reduce((s, k) => s + (k.difficulty || 50), 0) / Math.max(relData.length, 1)),
      intent: Object.keys(intentPcts).length > 0 ? intentPcts : { Informational: 42, Navigational: 20, Commercial: 20, Transactional: 14, Task: 4 },
      topBrands: brands,
      topics: topicsList
    });

  } catch (err) {
    console.error('Prompts API error:', err.message);
    res.status(500).json({ error: err.message, demo: true });
  }
});

// ROUTE 5: SITE AUDIT
// GET /api/audit?domain=example.com
// ═══════════════════════════════════════════════════════════════
app.get('/api/audit', async (req, res) => {
  const { domain } = req.query;
  if (!domain) return res.status(400).json({ error: 'domain is required' });

  try {
    const api = new DataForSEO();

    // Create on-page task
    const taskRes = await api.post('/v3/on_page/task_post', [{
      target: domain,
      max_crawl_pages: 100,
      user_agent_type: 'desktop',
      check_spelling: false,
      enable_javascript: false,
      custom_user_agent: 'GPTBot/1.0',
      store_raw_html: false
    }]);

    const taskId = taskRes?.tasks?.[0]?.id;
    if (!taskId) throw new Error('Failed to create audit task');

    // Poll for results (max 30s)
    let auditData = null;
    for (let i = 0; i < 6; i++) {
      await sleep(5000);
      const summaryRes = await api.post('/v3/on_page/summary', [{ id: taskId }]);
      const summary = summaryRes?.tasks?.[0]?.result?.[0];
      if (summary && summary.crawl_status === 'finished') {
        auditData = summary;
        break;
      }
    }

    if (!auditData) {
      // Return task ID for async polling
      return res.json({ taskId, status: 'processing', message: 'Audit in progress, check back in 1–2 minutes' });
    }

    const pages = auditData.page_metrics || {};
    const errors = pages.errors || 0;
    const warnings = pages.warnings || 0;
    const crawled = auditData.pages_crawled || 0;
    const healthy = crawled - errors - warnings;
    const siteHealth = Math.round(((crawled - errors) / Math.max(crawled, 1)) * 100);
    const aiHealth = auditData.robots_txt_accessible ? 95 : 40;

    // Get issues list
    const issuesRes = await api.post('/v3/on_page/issues_summary', [{ id: taskId }]);
    const issues = issuesRes?.tasks?.[0]?.result || [];

    res.json({
      domain,
      taskId,
      siteHealth,
      aiHealth,
      errors,
      warnings,
      crawled,
      healthy: Math.max(healthy, 0),
      issues: issues.slice(0, 15).map(issue => ({
        type: issue.issue_type || 'error',
        title: issue.title || 'Unknown issue',
        description: issue.description || '',
        count: issue.count || 0,
        isAI: issue.title?.toLowerCase().includes('ai') || issue.title?.toLowerCase().includes('robot')
      })),
      bots: [
        { name: 'GPTBot (OpenAI)', allowed: auditData.robots_txt_accessible !== false, icon: '🤖' },
        { name: 'Google-Extended', allowed: true, icon: '🔍' },
        { name: 'PerplexityBot', allowed: true, icon: '🔎' },
        { name: 'ClaudeBot (Anthropic)', allowed: true, icon: '✨' },
        { name: 'Amazonbot', allowed: false, icon: '📦' }
      ]
    });

  } catch (err) {
    console.error('Audit API error:', err.message);
    res.status(500).json({ error: err.message, demo: true });
  }
});

// UTILITY FUNCTIONS

function geoToLocation(geo) {
  const map = {
    worldwide: null,
    us: 2840,
    gb: 2826,
    de: 2276,
    in: 2356,
    au: 2036,
    ca: 2124,
    fr: 2250
  };
  return map[geo] || null;
}

function getPeriodStart(period) {
  const d = new Date();
  if (period === '1m') d.setMonth(d.getMonth() - 1);
  else if (period === '6m') d.setMonth(d.getMonth() - 6);
  else d.setFullYear(d.getFullYear() - 2);
  return d.toISOString().split('T')[0];
}

function getToday() {
  return new Date().toISOString().split('T')[0];
}

function calculateScore(metrics) {
  const cit = metrics.citations_count || 0;
  const men = metrics.mentions_count || 0;
  const pgs = metrics.cited_pages_count || 0;
  const score = Math.min(100, Math.round((cit * 0.5 + men * 0.3 + pgs * 0.2) / 10));
  return score || null;
}

function formatChange(val) {
  if (!val) return '';
  return val > 0 ? `↑${val}%` : `↓${Math.abs(val)}%`;
}

function formatNumber(n) {
  if (n >= 1000000) return (n / 1000000).toFixed(1) + 'M';
  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
  return String(n);
}

function buildTrend(metrics, period) {
  const labels = getLast6Months();
  return {
    labels,
    citations: labels.map((_, i) => Math.max(0, (metrics.citations_count || 0) - (labels.length - i - 1) * 5)),
    mentions: labels.map((_, i) => Math.max(0, (metrics.mentions_count || 0) - (labels.length - i - 1) * 2))
  };
}

function getLast6Months() {
  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  const result = [];
  const d = new Date();
  for (let i = 5; i >= 0; i--) {
    const m = new Date(d.getFullYear(), d.getMonth() - i, 1);
    result.push(months[m.getMonth()] + ' ' + String(m.getFullYear()).slice(2));
  }
  return result;
}
