app.post('/api/ai-pr/find-journalists', 
async (req, res) => {
  try {
    const { topic } = req.body;
    
    // Step 1: Find recent articles on topic
    // Use "news" category + auto type
    // NOTE: news category supports date filters
    const articleResults = await exa.search(
      `${topic} SEO AI search journalist article 2026`,
      {
        type: 'auto',
        numResults: 20,
        category: 'news',
        startPublishedDate: '2025-01-01',
        contents: {
          highlights: {
            numSentences: 2,
            highlightsPerUrl: 1
          }
        }
      }
    );
    
    // Step 2: Extract and deduplicate authors
    const authorMap = {};
    
    articleResults.results.forEach(result => {
      if (!result.author) return;
      
      const author = result.author.trim();
      if (!authorMap[author]) {
        authorMap[author] = {
          name: author,
          articles: [],
          publication: extractDomain(result.url),
          topics: new Set()
        };
      }
      
      authorMap[author].articles.push({
        title: result.title,
        url: result.url,
        date: result.publishedDate,
        snippet: result.highlights?.[0] || ''
      });
    });
    
    // Step 3: Find each journalist's profile
    // Use "people" category — NO date filters
    const journalists = await Promise.all(
      Object.values(authorMap)
        .slice(0, 10)
        .map(async (journalist) => {
          try {
            const profileResults = await exa.search(
              `${journalist.name} journalist writer ${journalist.publication}`,
              {
                type: 'auto',
                numResults: 3,
                category: 'people',
                // NOTE: no date filters for people category
                contents: {
                  highlights: {
                    numSentences: 2,
                    highlightsPerUrl: 1
                  }
                }
              }
            );
            
            // Extract Twitter/LinkedIn from highlights
            const profileText = profileResults.results
              .map(r => r.highlights?.join(' ') || '')
              .join(' ');
              
            const twitterMatch = profileText.match(
              /@([a-zA-Z0-9_]+)/
            );
            const linkedinMatch = profileResults.results
              .find(r => r.url?.includes('linkedin.com'));
            
            return {
              ...journalist,
              articles: journalist.articles,
              article_count: journalist.articles.length,
              twitter: twitterMatch 
                ? `@${twitterMatch[1]}` : null,
              linkedin_url: linkedinMatch?.url || null,
              profile_snippet: profileResults
                .results[0]?.highlights?.[0] || null
            };
          } catch (err) {
            // Return journalist without profile if search fails
            return {
              ...journalist,
              article_count: journalist.articles.length,
              twitter: null,
              linkedin_url: null
            };
          }
        })
    );
    
    res.json({ 
      success: true, 
      journalists: journalists.filter(j => j.articles.length > 0)
    });
    
  } catch (error) {
    console.error('Journalist finder error:', error);
    res.status(500).json({ error: error.message });
  }
});

// Helper to extract domain as publication name
const extractDomain = (url) => {
  try {
    const domain = new URL(url).hostname
      .replace('www.', '')
      .split('.')[0];
    return domain.charAt(0).toUpperCase() + domain.slice(1);
  } catch { return 'Unknown'; }
};

app.post('/api/ai-pr/monitor-coverage',
async (req, res) => {
  try {
    const { brand, competitor } = req.body;
    
    // Run 3 searches in parallel
    const [brandMentions, competitorMentions, 
           similarSites] = await Promise.all([
      
      // Search 1: Your brand mentions in news
      exa.search(
        `${brand} AI visibility GEO tool review`,
        {
          type: 'auto',
          numResults: 10,
          category: 'news',
          startPublishedDate: '2025-06-01',
          contents: {
            highlights: {
              numSentences: 3,
              highlightsPerUrl: 2
            }
          }
        }
      ),
      
      // Search 2: Competitor mentions
      exa.search(
        `${competitor} AI SEO tool review mention`,
        {
          type: 'auto',
          numResults: 10,
          category: 'news',
          startPublishedDate: '2025-06-01',
          contents: {
            highlights: {
              numSentences: 3,
              highlightsPerUrl: 2
            }
          }
        }
      ),
      
      // Search 3: findSimilar to find coverage opportunities
      // Sites similar to geoiqai.com that cover AI tools
      exa.findSimilar(
        'https://geoiqai.com',
        {
          numResults: 10,
          contents: {
            highlights: {
              numSentences: 2
            }
          }
        }
      )
    ]);
    
    // Check if brand mentioned in competitor articles
    const opportunityArticles = competitorMentions.results
      .filter(article => {
        const fullText = [
          article.title,
          ...(article.highlights || [])
        ].join(' ').toLowerCase();
        return !fullText.includes(brand.toLowerCase());
      })
      .map(article => ({
        ...article,
        opportunity: true,
        note: `${competitor} mentioned but not ${brand}`
      }));
    
    res.json({
      success: true,
      brand_mentions: brandMentions.results,
      competitor_mentions: competitorMentions.results,
      opportunities: opportunityArticles,
      similar_sites: similarSites.results
    });
    
  } catch (error) {
    console.error('Coverage monitor error:', error);
    res.status(500).json({ error: error.message });
  }
});

app.post('/api/ai-pr/generate-pitch',
async (req, res) => {
  try {
    const { 
      journalistName, 
      publication,
      articleUrl,
      pitchAngle,
      keyMessage 
    } = req.body;
    
    let recentWork = '';
    
    // Fetch journalist recent articles via Exa
    // Use "deep" type to get richer context
    if (articleUrl) {
      const articleContent = await exa.search(
        `${journalistName} ${publication} 
         recent articles AI SEO GEO`,
        {
          type: 'deep',
          numResults: 3,
          category: 'news',
          contents: {
            highlights: {
              numSentences: 3,
              highlightsPerUrl: 2
            }
          }
        }
      );
      
      recentWork = articleContent.results
        .map(r => `Title: ${r.title}\n` +
          `Excerpt: ${r.highlights?.join(' ')}`)
        .join('\n\n');
    }
    
    // Generate pitch with Claude
    const response = await anthropic.messages.create({
      model: 'claude-opus-4-5',
      max_tokens: 800,
      system: `You are a PR specialist for B2B SaaS.
Return ONLY raw JSON. No markdown. No backticks.`,
      messages: [{
        role: 'user',
        content: `Write a personalized journalist pitch email.

PRODUCT:
GeoIQ (geoiqai.com) tracks if brands appear 
in ChatGPT, Gemini, Perplexity, Claude, Grok,
and Google AI Overview. Gives GEO score and 
specific fixes. Free audit, paid from $69/mo.

JOURNALIST: ${journalistName}
PUBLICATION: ${publication}
PITCH ANGLE: ${pitchAngle}
KEY MESSAGE: ${keyMessage || 'not provided'}

THEIR RECENT WORK:
${recentWork || 'No articles fetched'}

Write email under 150 words.
Reference their specific work.
No em dashes. No bullet points. No bold.
Human founder voice not PR agency.

Return this JSON only:
{
  "subject": "Subject line under 60 chars",
  "body": "Full email body",
  "personalization": "Why this works for this journalist"
}`
      }]
    });
    
    const raw = response.content[0].text
      .replace(/\`\`\`json\n?/g, '')
      .replace(/\`\`\`\n?/g, '')
      .trim();
      
    const pitch = JSON.parse(raw);
    
    // Save to outreach tracker
    await supabase.from('pr_outreach').insert({
      user_id: req.user.id,
      journalist_name: journalistName,
      publication: publication,
      email_subject: pitch.subject,
      email_body: pitch.body,
      status: 'draft',
      created_at: new Date().toISOString()
    });
    
    res.json({ success: true, pitch });
    
  } catch (error) {
    console.error('Pitch generator error:', error);
    res.status(500).json({ error: error.message });
  }
});
