function resultMentionsBrand(
  text: string, 
  domain: string, 
  brandSlug: string
): boolean {
  const t = text.toLowerCase();
  const cleanDomain = domain
    .toLowerCase()
    .replace('www.','');
  
  // For very short slugs (1-2 chars)
  // require word boundary match
  if (brandSlug.length <= 2) {
    const wordBoundaryRegex = new RegExp(
      `\\b${brandSlug}\\b`, 'i'
    );
    return (
      wordBoundaryRegex.test(text) ||
      t.includes(cleanDomain)
    );
  }
  
  // For normal slugs use includes
  // but require domain OR full brand name
  return (
    t.includes(` ${brandSlug} `) ||    // space-padded
    t.includes(`"${brandSlug}"`) ||    // quoted
    t.startsWith(brandSlug) ||
    t.includes(cleanDomain)
  );
}

Also fix brandSlug extraction:

const getBrandSlug = (domain: string): string => {
  const clean = domain
    .replace('www.','')
    .toLowerCase();
  
  const name = clean.split('.')[0];
  
  // Short name (1-2 chars) = use full domain
  // to avoid false matches
  if (name.length <= 2) {
    return clean; // use "x.com" not "x"
  }
  
  return name; // "clevertap", "hubspot" etc
};

Fix — dynamic scoring based on evidence quality:

function scorePlatform(
  tavilyResults: TavilyResult[],
  domain: string,
  brandSlug: string,
  exaEvidence?: {
    mentionCount: number,
    hasDirectCitation: boolean,
    recentMentions: number
  }
): { found: boolean; score: number; evidence: string | null } {
  
  const cleanDomain = domain.replace('www.','');
  
  const matchingTavily = tavilyResults.filter(r =>
    resultMentionsBrand(
      `${r.title ?? ''} ${r.content ?? ''}`,
      domain,
      brandSlug
    )
  );
  
  // Base score from Tavily matches
  // More granular than before
  let baseScore = 0;
  if (matchingTavily.length === 0) baseScore = 0;
  else if (matchingTavily.length === 1) baseScore = 20;
  else if (matchingTavily.length === 2) baseScore = 35;
  else if (matchingTavily.length === 3) baseScore = 50;
  else if (matchingTavily.length === 4) baseScore = 65;
  else if (matchingTavily.length >= 5) baseScore = 75;
  
  // Quality bonuses (push above 75)
  let bonus = 0;
  
  // Bonus: result has domain in URL
  const hasDomainInUrl = matchingTavily
    .some(r => r.url?.includes(cleanDomain));
  if (hasDomainInUrl) bonus += 10;
  
  // Bonus: recent mentions (last 3 months)
  const recentResults = matchingTavily.filter(r => {
    if (!r.publishedDate) return false;
    const age = Date.now() - 
      new Date(r.publishedDate).getTime();
    return age < 90 * 24 * 60 * 60 * 1000;
  });
  if (recentResults.length > 0) bonus += 10;
  
  // Bonus: authoritative source
  const authoritativeDomains = [
    'techcrunch.com', 'wired.com',
    'theverge.com', 'searchengineland.com',
    'venturebeat.com', 'bloomberg.com',
    'reuters.com', 'forbes.com'
  ];
  const hasAuthoritativeSource = matchingTavily
    .some(r => authoritativeDomains
      .some(d => r.url?.includes(d))
    );
  if (hasAuthoritativeSource) bonus += 5;
  
  // Exa boost for very well known brands
  if (exaEvidence) {
    if (exaEvidence.hasDirectCitation) bonus += 10;
    if (exaEvidence.recentMentions > 5) bonus += 5;
  }
  
  const finalScore = Math.min(baseScore + bonus, 99);
  const found = finalScore >= 20;
  
  const evidence = matchingTavily[0]
    ? `${matchingTavily[0].title}: ${
        matchingTavily[0].content?.slice(0, 150)
      }`
    : null;
  
  return { found, score: finalScore, evidence };
}

const getPerPlatformExaEvidence = async (
  domain: string,
  brandSlug: string
) => {
  const cleanDomain = domain.replace('www.','');
  
  // Run platform-specific searches in parallel
  const [chatgptExa, geminiExa, perplexityExa] = 
    await Promise.allSettled([
      
      exa.search(
        `"${cleanDomain}" OR "${brandSlug}" 
         ChatGPT recommended suggest 2025 2026`,
        {
          type: 'auto',
          numResults: 5,
          startPublishedDate: '2025-06-01',
          contents: {
            highlights: {
              numSentences: 1
            }
          }
        }
      ),
      
      exa.search(
        `"${cleanDomain}" OR "${brandSlug}" 
         Gemini Google AI recommended 2025 2026`,
        {
          type: 'auto',
          numResults: 5,
          startPublishedDate: '2025-06-01',
          contents: {
            highlights: {
              numSentences: 1
            }
          }
        }
      ),
      
      exa.search(
        `"${cleanDomain}" OR "${brandSlug}" 
         Perplexity cited mentioned 2025 2026`,
        {
          type: 'auto',
          numResults: 5,
          startPublishedDate: '2025-06-01',
          contents: {
            highlights: {
              numSentences: 1
            }
          }
        }
      )
    ]);
  
  const extractEvidence = (result: any) => {
    if (result.status !== 'fulfilled') {
      return { mentionCount: 0, 
               hasDirectCitation: false,
               recentMentions: 0 };
    }
    const results = result.value.results || [];
    const mentioning = results.filter((r: any) =>
      resultMentionsBrand(
        `${r.title} ${r.highlights?.join(' ')}`,
        domain,
        brandSlug
      )
    );
    return {
      mentionCount: mentioning.length,
      hasDirectCitation: mentioning
        .some((r: any) => r.url?.includes(cleanDomain)),
      recentMentions: mentioning.filter((r: any) => {
        if (!r.publishedDate) return false;
        const age = Date.now() - 
          new Date(r.publishedDate).getTime();
        return age < 30 * 24 * 60 * 60 * 1000;
      }).length
    };
  };
  
  return {
    chatgpt: extractEvidence(chatgptExa),
    gemini: extractEvidence(geminiExa),
    perplexity: extractEvidence(perplexityExa)
  };
};
