const getGapImpact = (gapType, userRank, competitorRank) => {
  if (gapType === 'missing') return 'High';
  if (gapType === 'weak' && competitorRank <= 3) return 'High';

const CacheIndicator = ({ cachedAt }) => {
  const hoursAgo = Math.floor(
    (Date.now() - new Date(cachedAt)) / 3600000
  );
  const hoursLeft = Math.max(0, 72 - hoursAgo);
  
  return (
    <p style={{ 
      fontSize: 11, 
      color: '#9ca3af',
      marginTop: 8 
    }}>
      Data from {hoursAgo}h ago · 
      Refreshes in {hoursLeft}h
    </p>
  );
};
