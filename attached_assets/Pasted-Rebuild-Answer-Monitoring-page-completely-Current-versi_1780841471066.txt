Fix: auto-run all tracked prompts daily
using node-cron:

import cron from 'node-cron';

cron.schedule('0 6 * * *', async () => {
  console.log('Running daily prompt checks...');
  
  // Get all active tracked prompts
  const { data: prompts } = await supabase
    .from('tracked_prompts')
    .select('*, users(id, email)')
    .eq('active', true);
  
  for (const prompt of prompts) {
    try {
      // Check each LLM for this prompt
      for (const llm of prompt.llms) {
        await checkPrompt(
          prompt.user_id,
          prompt.domain,
          prompt.prompt,
          llm
        );
        // Wait 2 seconds between calls
        await new Promise(r => setTimeout(r, 2000));
      }
    } catch (err) {
      console.error(`Failed: ${prompt.prompt}`, err);
    }
  }
  
  console.log('Daily prompt checks complete');
});
