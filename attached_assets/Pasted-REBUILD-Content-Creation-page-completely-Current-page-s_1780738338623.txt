Source 1 — Prompt Research cache:

WHERE cache_key LIKE 'pr:%' 
AND cache_key LIKE '%' || domain || '%'
ORDER BY created_at DESC LIMIT 1

Source 2 — Competitor Research cache:
SELECT response FROM dataforseo_cache  
WHERE cache_key LIKE 'competitor:%'
AND cache_key LIKE '%' || domain || '%'
ORDER BY created_at DESC LIMIT 1

Source 3 — Brand Performance cache:
SELECT response FROM dataforseo_cache
WHERE cache_key LIKE 'brand_perf:%'
AND cache_key LIKE '%' || domain || '%'
ORDER BY created_at DESC LIMIT 1
