Group individual prompts/items into topic clusters.
Each item has se_query or question — cluster 
similar queries together.

Simple clustering by first 3 words:
const topicMap = {}

allItems.forEach(item => {
  const query = item.se_query || item.question || 
    item.keyword || ''
  
  // Simple topic key = first 3 words
  const words = query.toLowerCase().split(' ')
  const topicKey = words.slice(0, 3).join(' ')
  
  if (!topicMap[topicKey]) {
    topicMap[topicKey] = {
      topic: query,  // use first query as topic name
      prompts: [],
      totalAiVolume: 0,
      brands: new Set(),
      sources: new Set(),
      monthlySearches: []
    }
  }
  
  topicMap[topicKey].prompts.push(item)
  topicMap[topicKey].totalAiVolume += 
    item.ai_search_volume || 0
    
  // Collect brands from brand_entities
  const brandEntities = item.brand_entities || []
  brandEntities.forEach(b => {
    if (b.name) topicMap[topicKey].brands.add(b.name)
  })
  
  // Collect sources
  const sources = item.sources || []
  sources.forEach(s => {
    try {
      const domain = new URL(s.url).hostname
        .replace('www.','')
      topicMap[topicKey].sources.add(domain)
    } catch(e) {}
  })
  
  // Monthly trend data
  if (item.monthly_searches) {
    topicMap[topicKey].monthlySearches = 
      item.monthly_searches
  }
})

const topicsList = Object.values(topicMap)
  .sort((a,b) => b.totalAiVolume - a.totalAiVolume)
  .slice(0, 100)
