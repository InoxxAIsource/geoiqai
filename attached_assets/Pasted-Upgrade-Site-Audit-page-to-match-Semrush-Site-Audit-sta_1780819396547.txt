━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

app.post('/api/site-audit', async (req, res) => {
  const { domain } = req.body;
  const url = `https://${domain}`;
  const results = {};

  // Check 1: Server response + timing
  try {
    const start = Date.now();
    const resp = await fetch(url, { 
      timeout: 10000,
      headers: { 
        'User-Agent': 'Mozilla/5.0 GeoIQ-Audit/1.0' 
      }
    });
    results.server_response_ms = Date.now() - start;
    results.status_code = resp.status;
    results.https = url.startsWith('https');
    const html = await resp.text();
    
    // Check 2: Meta title
    const titleMatch = html.match(/<title[^>]*>(.*?)<\/title>/i);
    results.meta_title = titleMatch ? titleMatch[1].trim() : null;
    results.meta_title_length = results.meta_title?.length || 0;
    
    // Check 3: Meta description
    const descMatch = html.match(
      /<meta[^>]*name=["']description["'][^>]*content=["']([^"']*)["']/i
    );
    results.meta_description = descMatch ? descMatch[1] : null;
    results.meta_description_length = 
      results.meta_description?.length || 0;
    
    // Check 4: H1 tag
    const h1Match = html.match(/<h1[^>]*>(.*?)<\/h1>/i);
    results.has_h1 = !!h1Match;
    
    // Check 5: Schema markup
    results.has_schema = html.includes('application/ld+json');
    results.has_org_schema = html.includes('"Organization"');
    results.has_faq_schema = html.includes('"FAQPage"');
    results.has_software_schema = 
      html.includes('"SoftwareApplication"');
    
    // Check 6: Sitemap
    try {
      const sitemapResp = await fetch(`${url}/sitemap.xml`);
      results.has_sitemap = sitemapResp.status === 200;
    } catch { results.has_sitemap = false; }
    
    // Check 7: llms.txt
    try {
      const llmsResp = await fetch(`${url}/llms.txt`);
      results.has_llmstxt = llmsResp.status === 200;
    } catch { results.has_llmstxt = false; }
    
    // Check 8: robots.txt + bot access
    try {
      const robotsResp = await fetch(`${url}/robots.txt`);
      const robotsTxt = await robotsResp.text();
      results.robots_txt = robotsTxt;
      results.gptbot_allowed = !robotsTxt.includes('Disallow: /') 
        || robotsTxt.includes('GPTBot') && 
           !robotsTxt.match(/User-agent: GPTBot[\s\S]*?Disallow: \//);
      results.perplexitybot = 
        !robotsTxt.includes('PerplexityBot');
      results.claudebot = 
        !robotsTxt.includes('ClaudeBot');
      results.google_extended = 
        robotsTxt.includes('Google-Extended');
    } catch { 
      results.has_robots = false; 
    }
    
    // Check 9: Canonical
    results.has_canonical = html.includes('rel="canonical"');
    
    // Check 10: Image alts (basic check)
    const imgWithoutAlt = (html.match(/<img(?![^>]*alt=)[^>]*>/gi) || []).length;
    results.images_missing_alt = imgWithoutAlt;
    
  } catch (error) {
    results.fetch_error = error.message;
  }

  // Score calculation
  const checks = [
    results.meta_title,
    results.meta_description,
    results.has_h1,
    results.https,
    results.has_sitemap,
    results.has_schema,
    results.has_canonical,
    results.server_response_ms < 2000
  ];
  const passing = checks.filter(Boolean).length;
  results.site_health_score = Math.round((passing / checks.length) * 100);

  const aiChecks = [
    results.has_llmstxt,
    results.has_org_schema,
    results.has_faq_schema,
    results.gptbot_allowed,
    results.perplexitybot,
    results.claudebot,
    results.has_schema
  ];
  const aiPassing = aiChecks.filter(Boolean).length;
  results.ai_health_score = Math.round(
    (aiPassing / aiChecks.length) * 100
  );

  // Save to history
  await supabase.from('site_audit_history').insert({
    user_id: req.user.id,
    domain: domain,
    site_health_score: results.site_health_score,
    ai_health_score: results.ai_health_score,
    results: results,
    audited_at: new Date().toISOString()
  });

  res.json({ success: true, data: results });
});
