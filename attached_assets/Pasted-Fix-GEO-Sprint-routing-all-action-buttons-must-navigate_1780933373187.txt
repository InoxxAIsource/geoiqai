In AI Content Studio page component
on mount read sessionStorage:

useEffect(() => {
  const prefill = sessionStorage
    .getItem('sprint_prefill');
  
  if (prefill) {
    const data = JSON.parse(prefill);
    sessionStorage.removeItem('sprint_prefill');
    
    // Switch to correct tab
    if (data.tab) setActiveTab(data.tab);
    
    // Pre-fill content based on tab
    if (data.tab === 'geo-writer' && data.prompt) {
      setWriterPrompt(data.prompt);
      // Auto-trigger generation after 500ms
      setTimeout(() => triggerWriter(), 500);
    }
    
    if (data.tab === 'repurposer') {
      if (data.url) setRepurposerUrl(data.url);
      if (data.platform) {
        setSelectedPlatform(data.platform);
        // Auto-fetch URL content
        setTimeout(() => fetchUrl(), 500);
      }
    }
    
    if (data.tab === 'optimizer') {
      if (data.url) setOptimizerUrl(data.url);
      setTimeout(() => runAnalysis(), 500);
    }
  }
}, []);
