// Blocklist of common non-brand words
const NON_BRANDS = new Set([
  'the', 'a', 'an', 'some', 'top', 'best', 
  'new', 'free', 'video', 'streaming', 'service',
  'services', 'platform', 'app', 'content',
  'media', 'online', 'digital', 'watch',
  'movies', 'shows', 'series', 'episodes',
  'subscription', 'plan', 'price', 'cost',
  'how', 'what', 'why', 'when', 'where',
  'and', 'or', 'but', 'for', 'with', 'from'
])

// Only keep real brand names
function isRealBrand(name) {
  if (!name || name.length < 3) return false
  if (NON_BRANDS.has(name.toLowerCase())) return false
  if (/^\d+$/.test(name)) return false // pure numbers
  if (name.split(' ').length > 4) return false // too long
  return true
}

// Filter brands list
const cleanBrandsList = brandsList
  .filter(b => isRealBrand(b.name))
  .slice(0, 100)
