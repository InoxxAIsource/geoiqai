app.post('/api/generate/llmstxt', 
async (req, res) => {
  const { domain } = req.body;
  
  // Fetch domain content
  let content = '';
  try {
    const resp = await fetch(
      `https://${domain}`
    );
    const html = await resp.text();
    content = html
      .replace(/<[^>]+>/g,' ')
      .replace(/\s+/g,' ')
      .trim()
      .slice(0, 2000);
  } catch(e) {}
  
  const response = await anthropic
    .messages.create({
      model: 'claude-haiku-4-5-20251001',
      max_tokens: 800,
      system: 'Generate an llms.txt file. Return plain text only, no markdown.',
      messages: [{
        role: 'user',
        content: `Generate llms.txt for ${domain}.
                  
Content found: ${content}

Format:
# Brand Name - Tagline
> One paragraph description

## Key Topics
- Topic 1
- Topic 2

## Products/Services  
- Product 1
- Product 2

## Contact
- URL: https://${domain}

Keep under 500 words.
AI-optimized: clear, factual, structured.`
      }]
    });
  
  res.json({ 
    content: response.content[0].text,
    filename: 'llms.txt'
  });
});

app.post('/api/generate/org-schema',
async (req, res) => {
  const { domain } = req.body;
  
  const response = await anthropic
    .messages.create({
      model: 'claude-haiku-4-5-20251001',
      max_tokens: 600,
      system: 'Return only valid JSON-LD. No markdown.',
      messages: [{
        role: 'user',
        content: `Generate Organization schema 
for ${domain}.
Return complete JSON-LD script tag.
Include: name, url, description,
foundingDate if known, sameAs social links.`
      }]
    });
  
  res.json({ 
    code: response.content[0].text 
  });
});
