import Exa from 'exa-js';
import { tavily } from '@tavily/core';

const exa = new Exa(process.env.EXA_API_KEY);
const tvly = tavily({ 
  apiKey: process.env.TAVILY_API_KEY 
});

const runAIPresenceScan = async (domain) => {
  
  const brandName = domain
    .replace('www.','')
    .split('.')[0];
  
  // Run 4 searches in parallel
  // Total cost: ~$0.05
  const [
    aiCitations,
    chatgptEvidence,
    geminiEvidence, 
    perplexityEvidence,
    companyProfile
  ] = await Promise.allSettled([

    // Signal 1: Exa deep search with structured output
    // Finds real evidence of brand in AI context
    // Cost: ~$0.02 (deep type)
    exa.search(
      `${brandName} ${domain} recommended 
       mentioned by AI ChatGPT Gemini Perplexity`,
      {
        type: 'deep',
        numResults: 10,
        startPublishedDate: '2025-01-01',
        contents: {
          highlights: {
            numSentences: 2,
            highlightsPerUrl: 1
          }
        },
        output_schema: {
          type: 'object',
          properties: {
            mentioned_in_ai_context: {
              type: 'boolean'
            },
            visibility_strength: {
              type: 'string',
              enum: ['high','medium','low','none']
            },
            ai_systems_that_know_brand: {
              type: 'array',
              items: { type: 'string' }
            },
            evidence_count: { type: 'number' },
            top_evidence: { type: 'string' }
          }
        }
      }
    ),

    // Signal 2: Tavily — ChatGPT specific evidence
    // Cost: ~$0.008
    tvly.search(
      `${brandName} ChatGPT recommended suggest`,
      {
        searchDepth: 'basic',
        maxResults: 5,
        topic: 'general'
      }
    ),

    // Signal 3: Tavily — Gemini specific evidence
    // Cost: ~$0.008
    tvly.search(
      `${brandName} Gemini Google AI recommended`,
      {
        searchDepth: 'basic',
        maxResults: 5
      }
    ),

    // Signal 4: Tavily — Perplexity specific evidence
    // Cost: ~$0.008
    tvly.search(
      `${brandName} Perplexity cited source`,
      {
        searchDepth: 'basic',
        maxResults: 5
      }
    ),

    // Signal 5: Exa company profile
    // Gets brand description and reach
    // Cost: ~$0.01
    exa.search(
      `${domain} company brand`,
      {
        type: 'auto',
        numResults: 3,
        category: 'company',
        contents: {
          highlights: {
            numSentences: 2
          }
        }
      }
    )
  ]);

  // Extract results safely
  const aiData = aiCitations.status === 'fulfilled'
    ? aiCitations.value : null;
    
  const chatgptResults = chatgptEvidence.status 
    === 'fulfilled'
    ? chatgptEvidence.value.results : [];
    
  const geminiResults = geminiEvidence.status 
    === 'fulfilled'
    ? geminiEvidence.value.results : [];
    
  const perplexityResults = perplexityEvidence
    .status === 'fulfilled'
    ? perplexityEvidence.value.results : [];
    
  const companyData = companyProfile.status 
    === 'fulfilled'
    ? companyProfile.value.results : [];

  // Calculate per-system scores
  // Based on real evidence found
  
  const calcSystemScore = (
    results, 
    systemName,
    brandName
  ) => {
    if (!results || results.length === 0) {
      return { 
        mentioned: false, 
        score: 0, 
        evidence: null 
      };
    }
    
    // Check if brand actually mentioned
    const brandMentioned = results.some(r => 
      r.title?.toLowerCase()
        .includes(brandName.toLowerCase()) ||
      r.content?.toLowerCase()
        .includes(brandName.toLowerCase())
    );
    
    const relevantResults = results.filter(r =>
      r.content?.toLowerCase()
        .includes(brandName.toLowerCase())
    );
    
    const score = brandMentioned 
      ? Math.min(relevantResults.length * 20, 100)
      : 0;
      
    return {
      mentioned: brandMentioned,
      score,
      position: brandMentioned ? 1 : null,
      evidence: relevantResults[0]?.content
        ?.slice(0, 200) || null,
      url_cited: relevantResults.some(r => 
        r.url?.includes(domain)
      )
    };
  };

  // Build per-system results
  const systemResults = {
    chatgpt: calcSystemScore(
      chatgptResults, 'ChatGPT', brandName
    ),
    gemini: calcSystemScore(
      geminiResults, 'Gemini', brandName
    ),
    perplexity: calcSystemScore(
      perplexityResults, 'Perplexity', brandName
    ),
    // Claude and Grok use Exa AI citations data
    claude: {
      mentioned: aiData?.mentioned_in_ai_context 
        || false,
      score: aiData?.visibility_strength === 'high' 
        ? 80 
        : aiData?.visibility_strength === 'medium'
        ? 50
        : aiData?.visibility_strength === 'low'
        ? 20
        : 0,
      evidence: aiData?.top_evidence || null
    },
    grok: {
      mentioned: aiData?.evidence_count > 5,
      score: aiData?.evidence_count > 5 ? 60
        : aiData?.evidence_count > 2 ? 30
        : 0
    },
    google_ai: calcSystemScore(
      [...chatgptResults, ...geminiResults],
      'Google AI',
      brandName
    )
  };

  // Calculate overall metrics
  const systemsFound = Object.values(systemResults)
    .filter(s => s.mentioned).length;
    
  const overallScore = Math.round(
    Object.values(systemResults)
      .reduce((sum, s) => sum + (s.score || 0), 0) 
    / 6
  );

  const urlsCited = Object.values(systemResults)
    .filter(s => s.url_cited).length;

  return {
    score: overallScore,
    label: overallScore >= 70 ? 'Strong'
      : overallScore >= 40 ? 'Moderate'
      : overallScore >= 20 ? 'Low'
      : 'Not Visible',
    
    // New metrics (replace DataForSEO fields)
    ai_presence_pct: overallScore,
    systems_found: systemsFound,
    urls_cited: urlsCited,
    cited_sources: companyData.length,
    
    // Per system breakdown
    systems: systemResults,
    
    // Company context
    company_description: companyData[0]
      ?.highlights?.[0] || null,
    
    // Evidence summary
    evidence_count: aiData?.evidence_count || 0,
    top_evidence: aiData?.top_evidence || null
  };
};
