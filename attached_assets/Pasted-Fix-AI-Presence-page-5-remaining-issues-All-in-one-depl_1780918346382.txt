const detectPlatformFromUrl = (url) => {
  if (!url) return 'Web';
  if (url.includes('reddit.com')) 
    return 'Reddit';
  if (url.includes('twitter.com') || 
      url.includes('x.com')) 
    return 'X/Twitter';
  if (url.includes('linkedin.com')) 
    return 'LinkedIn';
  if (url.includes('medium.com')) 
    return 'Medium';
  if (url.includes('youtube.com')) 
    return 'YouTube';
  return 'Web';
};

const getGoogleAIOScore = (aioData) => {
  if (!aioData) return 0;
  if (aioData.cited === true) {
    const queryBonus = Math.min(
      (aioData.queries?.length || 1) * 5, 
      19
    );
    return 80 + queryBonus;
  }
  if (aioData.mentioned === true) return 45;
  return 10;
};
