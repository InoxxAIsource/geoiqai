const validateDomain = async (domain) => {
  const clean = domain
    .replace('www.','')
    .replace('https://','')
    .replace('http://','')
    .toLowerCase()
    .trim();
  
  // Basic format check
  if (!clean.includes('.') || 
      clean.length < 4) {
    return { 
      valid: false, 
      error: 'Invalid domain format' 
    };
  }
  
  // Check domain resolves
  try {
    const resp = await fetch(
      `https://${clean}`, {
        method: 'HEAD',
        signal: AbortSignal.timeout(5000)
      }
    );
    return { valid: true, domain: clean };
  } catch {
    // Try without HEAD
    try {
      const resp = await fetch(
        `https://${clean}`, {
          signal: AbortSignal.timeout(5000)
        }
      );
      return { valid: true, domain: clean };
    } catch {
      return { 
        valid: false, 
        error: `Cannot reach ${clean}. Check spelling.`
      };
    }
  }
};

const getCompetitorScore = async (domain) => {
  console.log(`Scoring ${domain}...`);
  
  const cleanDomain = domain
    .replace('www.','').toLowerCase();
  const brandName = extractBrandName(domain);
  
  console.log(`Brand slug: ${brandName}`);
  
  // Exa search
  let exaResults = [];
  try {
    const exaResp = await exa.search(
      `"${brandName}" OR "${cleanDomain}" 
       recommended mentioned AI 2025 2026`,
      {
        type: 'auto',
        numResults: 15,
        category: 'news',
        startPublishedDate: '2025-01-01',
        contents: {
          highlights: {
            numSentences: 1,
            highlightsPerUrl: 1
          }
        }
      }
    );
    exaResults = exaResp.results || [];
    console.log(`Exa results for ${domain}:`, 
      exaResults.length);

    console.error('Exa error:', e.message);
  }
  
  // Known tier baseline

  };
  
  // If known brand use tier score

  if (KNOWN_TIERS[cleanDomain]) {
    const baseScore = KNOWN_TIERS[cleanDomain];
    const exaBonus = Math.min(
      exaResults.length * 2, 10
    );
    return Math.min(baseScore + exaBonus, 99);
  }
  
  // Unknown brand: calculate from Exa
  if (exaResults.length === 0) {
    // Brand too new/small for web evidence
    // Use Claude to estimate
    const claudeResp = await anthropic

Perplexity mention this brand?
Consider: brand size, industry presence,
how well known globally.
Return only the number.`
        }]
      });
    
    const score = parseInt(
      claudeResp.content[0].text.trim()
    ) || 10;
    return Math.min(score, 99);
  }
  
  // Calculate from Exa results
  const presenceScore = Math.min(
    exaResults.length * 5, 60
  );
  const freshnessBonus = exaResults
    .filter(r => {
      const age = Date.now() - 
        new Date(r.publishedDate).getTime();
      return age < 30 * 24 * 60 * 60 * 1000;
    }).length * 3;
  
  return Math.min(
    presenceScore + freshnessBonus, 
    99
  );

const getTopicGaps = async (
  userDomain, 
  competitorDomain
) => {
  const userBrand = extractBrandName(userDomain);
  const compBrand = extractBrandName(
    competitorDomain
  );
  
  console.log(
    `Getting topics: ${userBrand} vs ${compBrand}`
  );
  
  try {
    // Search for both brands in parallel
    const [userResults, compResults] = 
      await Promise.all([
        exa.search(
          `"${userBrand}" OR "${userDomain
            .replace('www.','')}" 
           software tool platform recommended 
           use case 2025 2026`,
          {
            type: 'auto',
            numResults: 15,
            category: 'news',
            startPublishedDate: '2025-01-01',
            contents: {
              highlights: {
                numSentences: 1
              }
            }
          }
        ),
        exa.search(
          `"${compBrand}" OR "${competitorDomain
            .replace('www.','')}" 
           software tool platform recommended 
           use case 2025 2026`,
          {
            type: 'auto',
            numResults: 15,
            category: 'news',
            startPublishedDate: '2025-01-01',
            contents: {
              highlights: {
                numSentences: 1
              }
            }
          }
        )
      ]);
    
    console.log(
      `User results: ${userResults.results.length}`
    );
    console.log(
      `Comp results: ${compResults.results.length}`
    );
    
    // Extract topics from titles
    const userTopics = new Set(
      userResults.results
        .map(r => r.title
          ?.replace(/[|\-–—].*$/, '')
          .trim()
          .slice(0, 60)
        )
        .filter(t => t?.length > 10)
    );
    
    const compTopics = new Set(
      compResults.results
        .map(r => r.title
          ?.replace(/[|\-–—].*$/, '')
          .trim()
          .slice(0, 60)
        )
        .filter(t => t?.length > 10)
    );
    
    // Send to Claude for gap analysis
    const gapResp = await anthropic
      .messages.create({
        model: 'claude-haiku-4-5-20251001',
        max_tokens: 800,
        system: 'Return only raw JSON. No markdown.',
        messages: [{
          role: 'user',
          content: `Analyze AI visibility topic gaps.

${userDomain} appears in these topics:
${[...userTopics].join('\n')}

${competitorDomain} appears in these topics:
${[...compTopics].join('\n')}

Compare and classify each unique topic.
Return JSON:
{
  "topics": [
    {
      "topic": "topic name (max 50 chars)",
      "user_present": true/false,
      "competitor_present": true/false,
      "gap_type": "missing|weak|shared|strong|unique",
      "impact": "High|Medium|Low",
      "action": "specific action to take"
    }
  ],
  "summary": {
    "missing": 0,
    "weak": 0,
    "shared": 0,
    "strong": 0,
    "unique": 0
  }
}`
        }]
      });
    
    const parsed = JSON.parse(
      gapResp.content[0].text
        .replace(/\`\`\`json\n?/g, '')
        .replace(/\`\`\`\n?/g, '')
        .trim()
    );
    
    console.log(
      'Topics found:', 
      parsed.topics?.length
    );
    return parsed;
    
  } catch (err) {
    console.error('Topic gap error:', err);
    return { topics: [], summary: {} };
  }
};
