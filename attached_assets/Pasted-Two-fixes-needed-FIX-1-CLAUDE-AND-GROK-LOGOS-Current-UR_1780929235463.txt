const parseAIOResult = (serpData) => {
  try {
    const task = serpData.tasks?.[0];
    if (!task) return null;
    
    const result = task.result?.[0];
    if (!result) return null;
    
    const items = result.items || [];
    
    // Check every possible location
    
    // Location 1: direct ai_overview type
    const aioItem = items.find(i => 
      i.type === 'ai_overview'
    );
    if (aioItem) {
      console.log('Found AIO at type ai_overview');
      return aioItem;
    }
    
    // Location 2: featured_snippet
    const featuredSnippet = items.find(i =>
      i.type === 'featured_snippet'
    );
    if (featuredSnippet) {
      console.log('Found featured_snippet');
      return featuredSnippet;
    }
    
    // Location 3: answer_box
    const answerBox = items.find(i =>
      i.type === 'answer_box'
    );
    if (answerBox) {
      console.log('Found answer_box');
      return answerBox;
    }
    
    // Location 4: direct on result
    if (result.ai_overview) {
      console.log('Found on result.ai_overview');
      return result.ai_overview;
    }
    
    // Location 5: check task extras
    if (task.data?.ai_overview) {
      console.log('Found on task.data');
      return task.data.ai_overview;
    }
    
    console.log('No AIO found. Item types:', 
      items.map(i => i.type)
    );
    return null;
    
  } catch (err) {
    console.error('AIO parse error:', err);
    return null;
  }
};
