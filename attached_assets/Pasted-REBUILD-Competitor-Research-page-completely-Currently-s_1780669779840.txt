Extract topic/question from each item:
const yourTopicMap = {}
yourTopics.tasks[0].result[0]?.items?.forEach(item => {
  const topic = item.se_query || item.question || item.keyword || ''
  if (topic) yourTopicMap[topic.toLowerCase()] = {
    topic,
    yourMentions: 1,
    aiVolume: item.ai_search_volume || 0
  }
})

const mergedTopics = {}

// Add your topics
Object.entries(yourTopicMap).forEach(([key, data]) => {
  mergedTopics[key] = { 
    topic: data.topic,
    yourMentions: data.yourMentions,
    competitorMentions: 0,
    aiVolume: data.aiVolume
  }
})

// Add competitor topics
competitorTopics.tasks[0].result[0]?.items?.forEach(item => {
  const topic = (item.se_query || item.question || item.keyword || '').toLowerCase()
  if (!topic) return
  if (mergedTopics[topic]) {
    mergedTopics[topic].competitorMentions = 1
  } else {
    mergedTopics[topic] = {
      topic: item.se_query || item.question || item.keyword,
      yourMentions: 0,
      competitorMentions: 1,
      aiVolume: item.ai_search_volume || 0
    }
  }
})

// Classify each topic
const topicList = Object.values(mergedTopics).map(t => ({
  ...t,
  status: t.yourMentions > 0 && t.competitorMentions === 0 ? 'unique' :
          t.yourMentions === 0 && t.competitorMentions > 0 ? 'missing' :
          t.yourMentions > 0 && t.competitorMentions > 0 ? 'shared' : 'unknown'
}))

const counts = {
  all: topicList.length,
  missing: topicList.filter(t => t.status === 'missing').length,
  shared: topicList.filter(t => t.status === 'shared').length,
  unique: topicList.filter(t => t.status === 'unique').length
}
