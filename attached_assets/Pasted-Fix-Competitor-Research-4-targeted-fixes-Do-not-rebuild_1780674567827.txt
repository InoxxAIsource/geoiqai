// utils/scoreCalculator.js
export function calculateVisibilityScore(
  mentions, citations = 0, citedPages = 0
) {
  const mentionScore = Math.min(
    Math.log10(Math.max(mentions, 1)) / 
    Math.log10(10000000) * 70, 70
  )
  const citationBonus = citations > 0
    ? Math.min(Math.log10(citations) / 6 * 20, 20)
    : 0
  const pageBonus = citedPages > 0
    ? Math.min(citedPages / 100 * 10, 10)
    : 0
  return Math.min(100, Math.round(
    mentionScore + citationBonus + pageBonus
  ))
}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Fix classification to compare correctly:

// yourCount and competitorCount should be 
// based on whether topic appears in each 
// brand's search results, weighted by position

topics.forEach((topic, index) => {
  const yourRank = yourTopicMap[topic.key]?.rank || 999
  const compRank = competitorTopicMap[topic.key]?.rank || 999
  
  // Lower rank = more prominent
  if (yourRank < 999 && compRank === 999) {
    topic.status = 'unique'
  } else if (yourRank === 999 && compRank < 999) {
    topic.status = 'missing'
  } else if (yourRank < compRank * 0.5) {
    topic.status = 'strong'  // you rank much better
  } else if (compRank < yourRank * 0.5) {
    topic.status = 'weak'    // competitor ranks much better
  } else {
    topic.status = 'shared'  // roughly equal
  }
})
