CREATE TABLE IF NOT EXISTS sprint_progress (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id uuid NOT NULL,
  domain text NOT NULL,
  step_id text NOT NULL,
  phase integer NOT NULL,
  completed boolean DEFAULT false,
  completed_at timestamptz,
  score_before integer DEFAULT 0,
  score_after integer DEFAULT 0,
  notes text,
  created_at timestamptz DEFAULT now(),
  UNIQUE(user_id, domain, step_id)
);

CREATE TABLE IF NOT EXISTS sprint_sessions (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id uuid NOT NULL,
  domain text NOT NULL,
  started_at timestamptz DEFAULT now(),
  target_score integer DEFAULT 30,
  current_score integer DEFAULT 0,
  last_updated timestamptz DEFAULT now()
);

CREATE INDEX idx_sprint_user_domain 
  ON sprint_progress(user_id, domain);

app.get('/api/sprint/:domain', 
async (req, res) => {
  const { domain } = req.params;
  const userId = req.user.id;
  
  // Get or create sprint session
  let { data: session } = await supabase
    .from('sprint_sessions')
    .select('*')
    .eq('user_id', userId)
    .eq('domain', domain)
    .single();
  
  if (!session) {
    // Get current AI presence score
    const { data: latestScan } = await supabase
      .from('geoiq_cache')
      .select('response_json')
      .eq('domain', domain)
      .eq('endpoint', 'ai_presence')
      .order('cached_at', { ascending: false })
      .limit(1)
      .single();
    
    const currentScore = latestScan
      ?.response_json?.score || 0;
    
    const { data: newSession } = await supabase
      .from('sprint_sessions')
      .insert({
        user_id: userId,
        domain,
        target_score: 30,
        current_score: currentScore,
        started_at: new Date().toISOString(),
        last_updated: new Date().toISOString()
      })
      .select()
      .single();
    
    session = newSession;
  }
  
  // Get completed steps
  const { data: completedSteps } = await supabase
    .from('sprint_progress')
    .select('step_id, completed_at, score_after')
    .eq('user_id', userId)
    .eq('domain', domain)
    .eq('completed', true);
  
  const completedIds = new Set(
    completedSteps?.map(s => s.step_id) || []
  );
  
  // Calculate projected score
  const completedPts = SPRINT_STEPS
    .filter(s => completedIds.has(s.id))
    .reduce((sum, s) => sum + s.score_impact, 0);
  
  const projectedScore = Math.min(
    (session.current_score || 0) + completedPts,
    99
  );
  
  res.json({
    session,
    steps: SPRINT_STEPS.map(s => ({
      ...s,
      completed: completedIds.has(s.id),
      completed_at: completedSteps
        ?.find(c => c.step_id === s.id)
        ?.completed_at
    })),
    stats: {
      total_steps: SPRINT_STEPS.length,
      completed_count: completedIds.size,
      current_score: session.current_score,
      projected_score: projectedScore,
      target_score: 30,
      pts_remaining: Math.max(
        30 - projectedScore, 0
      )
    }
  });
});

app.post('/api/sprint/complete-step',
async (req, res) => {
  const { domain, step_id } = req.body;
  const userId = req.user.id;
  
  // Get current score
  const { data: session } = await supabase
    .from('sprint_sessions')
    .select('current_score')
    .eq('user_id', userId)
    .eq('domain', domain)
    .single();
  
  const step = SPRINT_STEPS.find(
    s => s.id === step_id
  );
  
  await supabase
    .from('sprint_progress')
    .upsert({
      user_id: userId,
      domain,
      step_id,
      phase: step?.phase || 1,
      completed: true,
      completed_at: new Date().toISOString(),
      score_before: session?.current_score || 0,
      score_after: (session?.current_score || 0) 
        + (step?.score_impact || 0)
    }, { onConflict: 'user_id,domain,step_id' });
  
  res.json({ success: true });
});
