At the VERY TOP of the competitor

DataForSEO function is called:

app.post('/api/dataforseo/competitor-research',
async (req, res) => {
  const { domain, competitors } = req.body;
  
  // BUILD CACHE KEY
  const competitorKey = competitors
    .slice()
    .sort()
    .join(',');
  const cacheKey = `competitor::${domain
    .toLowerCase()
    .replace('www.','')}::${competitorKey}`;
  
  // CHECK CACHE FIRST — BEFORE ANYTHING ELSE
  const { data: cached } = await supabase
    .from('dataforseo_cache')
    .select('response_data, created_at')
    .eq('cache_key', cacheKey)
    .gt('created_at', 
      new Date(Date.now() - 72 * 60 * 60 * 1000)
        .toISOString()
    )
    .single();
  
  if (cached) {
    console.log(
      `[CACHE HIT] competitor ${domain} 
       — NO DataForSEO calls fired`
    );
    return res.json({
      success: true,
      data: cached.response_data,
      from_cache: true,
      cached_at: cached.created_at
    });
  }
  
  console.log(
    `[CACHE MISS] competitor ${domain} 
     — calling DataForSEO`
  );
  
  // Only reaches here on cache miss
  // Run actual DataForSEO calls...
});

  competitors.map(c => fetchCompetitorData(c))

let callCount = 0;
const MAX_CALLS_PER_REQUEST = 20;

const safeDataForSEOCall = async (fn) => {
  callCount++;
  if (callCount > MAX_CALLS_PER_REQUEST) {
    console.error(
      `[ABORT] DataForSEO call limit reached: 
       ${callCount} calls attempted`
    );
    throw new Error(
      'Too many API calls in one request'
    );
  }
  return await fn();
};
