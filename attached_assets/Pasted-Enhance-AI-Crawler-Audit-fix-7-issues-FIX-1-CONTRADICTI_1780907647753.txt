const getPageSpeedData = async (url) => {
  if (!process.env.GOOGLE_PSI_API_KEY) {
    return null;
  }
  
  try {
    const apiUrl = new URL(
      'https://www.googleapis.com/pagespeedonline/v5/runPagespeed'
    );
    apiUrl.searchParams.set('url', 
      `https://${url}`
    );
    apiUrl.searchParams.set('key', 
      process.env.GOOGLE_PSI_API_KEY
    );
    apiUrl.searchParams.set('strategy', 'mobile');
    
    const response = await fetch(apiUrl.toString());
    const data = await response.json();
    
    return {
      performance: Math.round(
        data.lighthouseResult
          .categories.performance.score * 100
      ),
      lcp: data.lighthouseResult.audits[
        'largest-contentful-paint'
      ].displayValue,
      cls: data.lighthouseResult.audits[
        'cumulative-layout-shift'
      ].displayValue,
      fcp: data.lighthouseResult.audits[
        'first-contentful-paint'
      ].displayValue,
      ttfb: data.lighthouseResult.audits[
        'server-response-time'
      ].displayValue
    };
  } catch (err) {
    console.log('PSI API error:', err.message);
    return null;
  }
};

import { tavily } from '@tavily/core';
const tvly = tavily({ 
  apiKey: process.env.TAVILY_API_KEY 
});

const crawlSite = async (domain) => {
  try {
    const result = await tvly.crawl(
      `https://${domain}`,
      {
        maxDepth: 2,
        maxPages: 10,
        extractContent: true
      }
    );
    
    return result.pages.map(page => ({
      url: page.url,
      statusCode: page.statusCode || 200,
      title: page.title,
      content: page.content,
      hasH1: page.content?.includes('<h1'),
      wordCount: page.content?.split(' ').length
    }));
    
  } catch (err) {
    console.log('Tavily crawl failed, 
      falling back to direct fetch');
    return null;
  }
};
