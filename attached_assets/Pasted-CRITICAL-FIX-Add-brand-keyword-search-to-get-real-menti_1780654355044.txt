When user enters "netflix.com", extract brand name:

function extractBrandName(domain) {
  // Remove TLD and www
  // "netflix.com" → "netflix"
  // "hubspot.com" → "hubspot"
  // "geoiqai.com" → "geoiqai"
  // "coca-cola.com" → "coca-cola"
  return domain
    .replace(/^www\./, '')
    .replace(/\.[a-z]{2,}(\.[a-z]{2})?$/, '')
    .toLowerCase()
    .trim()
}
