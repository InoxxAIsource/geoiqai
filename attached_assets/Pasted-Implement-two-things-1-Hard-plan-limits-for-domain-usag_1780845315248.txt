GET USER PLAN FROM SUPABASE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

const getUserPlan = async (userId) => {
  const { data } = await supabase
    .from('subscriptions')
    .select('plan, status')
    .eq('user_id', userId)
    .eq('status', 'active')
    .single();
  if (!data) return 'free';
  return data.plan;
};

const getUserLimits = async (userId) => {
  const plan = await getUserPlan(userId);
  return { plan, limits: PLAN_LIMITS[plan] };
};

const checkDomainLimit = async (
  userId,
  featureType
) => {
  const { limits } = await getUserLimits(userId);
  const limitKey = `${featureType}_domains`;
  const maxDomains = limits[limitKey];

  const { count } = await supabase
    .from('user_domains')
    .select('*', { count: 'exact' })
    .eq('user_id', userId)
    .eq(`${featureType}_enabled`, true);

  if (count >= maxDomains) {
    return {
      allowed: false,
      current: count,
      max: maxDomains,
      upgrade_required: true
    };
  }
  return { 
    allowed: true, 
    current: count, 
    max: maxDomains 
  };
};

const checkRescanAllowed = async (
  userId,
  domain,
  endpoint
) => {
  const { limits } = await getUserLimits(userId);
  const minHours = limits.rescan_hours;

  const { data: lastScan } = await supabase
    .from('geoiq_cache')
    .select('cached_at')
    .eq('domain', domain)
    .eq('endpoint', endpoint)
    .order('cached_at', { ascending: false })
    .limit(1)
    .single();

  if (!lastScan) return { allowed: true };

  const hoursSince = (
    Date.now() - new Date(lastScan.cached_at)
  ) / (1000 * 60 * 60);

  if (hoursSince < minHours) {
    const hoursLeft = Math.ceil(
      minHours - hoursSince
    );
    return {
      allowed: false,
      hours_left: hoursLeft,
      message: `Next scan available in ${hoursLeft} hours`
    };
  }

  return { allowed: true };
};

SUPABASE CACHE TABLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CREATE TABLE geoiq_cache (
  id uuid DEFAULT gen_random_uuid(),
  cache_key text UNIQUE NOT NULL,
  endpoint text NOT NULL,
  domain text,
  user_id uuid,
  response_json jsonb NOT NULL,
  cached_at timestamptz DEFAULT now(),
  expires_at timestamptz NOT NULL,
  hit_count integer DEFAULT 0,
  last_hit_at timestamptz
);

CREATE INDEX idx_cache_key 
  ON geoiq_cache(cache_key);
CREATE INDEX idx_cache_domain 
  ON geoiq_cache(domain);
CREATE INDEX idx_cache_expires 
  ON geoiq_cache(expires_at);

CACHE KEY FORMAT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

const buildCacheKey = (
  endpoint, 
  domain, 
  params = {}
) => {
  const base = `${endpoint}::${domain
    .toLowerCase()
    .replace('www.', '')
    .trim()}`;

  if (Object.keys(params).length === 0) 
    return base;

  const paramStr = Object.entries(params)
    .sort(([a], [b]) => a.localeCompare(b))
    .map(([k, v]) => `${k}=${v}`)
    .join('::');

  return `${base}::${paramStr}`;
};

CACHE UTILITY FUNCTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

const getCache = async (cacheKey) => {
  try {
    const { data, error } = await supabase
      .from('geoiq_cache')
      .select(
        'response_json, cached_at, expires_at, hit_count'
      )
      .eq('cache_key', cacheKey)
      .gt('expires_at', new Date().toISOString())
      .single();

    if (error || !data) return null;

    // Update hit count silently
    supabase
      .from('geoiq_cache')
      .update({
        hit_count: data.hit_count + 1,
        last_hit_at: new Date().toISOString()
      })
      .eq('cache_key', cacheKey)
      .then(() => {});

    return {
      data: data.response_json,
      cached_at: data.cached_at,
      expires_at: data.expires_at,
      from_cache: true
    };
  } catch (err) {
    console.error('Cache read error:', err);
    return null;
  }
};

const setCache = async (
  cacheKey,
  endpoint,
  domain,
  responseData,
  userId = null
) => {
  try {
    const duration = 
      CACHE_DURATIONS[endpoint] || 
      24 * 60 * 60 * 1000;

    const expiresAt = new Date(
      Date.now() + duration
    ).toISOString();

    await supabase
      .from('geoiq_cache')
      .upsert({
        cache_key: cacheKey,
        endpoint,
        domain: domain
          ?.replace('www.', '')
          .toLowerCase(),
        user_id: userId,
        response_json: responseData,
        cached_at: new Date().toISOString(),
        expires_at: expiresAt,
        hit_count: 0
      }, {
        onConflict: 'cache_key'
      });

    console.log(
      `Cached ${endpoint} for ${domain} 
       until ${expiresAt}`
    );
  } catch (err) {
    console.error('Cache write error:', err);
  }
};

const clearCache = async (
  domain, 
  endpoint = null
) => {
  const query = supabase
    .from('geoiq_cache')
    .delete()
    .eq('domain', domain
      ?.replace('www.', '')
      .toLowerCase()
    );

  if (endpoint) query.eq('endpoint', endpoint);
  await query;
};

app.post('/api/llm-mentions', 
async (req, res) => {
  try {
    const { domain } = req.body;
    const cacheKey = buildCacheKey(
      'llm_mentions', domain
    );

    // CHECK CACHE FIRST
    const cached = await getCache(cacheKey);
    if (cached) {
      console.log(
        `Cache HIT: llm_mentions ${domain}`
      );
      return res.json({
        success: true,
        data: cached.data,
        from_cache: true,
        cached_at: cached.cached_at,
        expires_at: cached.expires_at
      });
    }

    console.log(
      `Cache MISS: llm_mentions ${domain} 
       calling API`
    );

    // CALL REAL API
    const result = await runLLMMentionsScan(domain);

    // STORE IN CACHE
    await setCache(
      cacheKey,
      'llm_mentions',
      domain,
      result,
      req.user?.id
    );

    res.json({
      success: true,
      data: result,
      from_cache: false,
      cached_at: new Date().toISOString(),
      expires_at: new Date(
        Date.now() + 72 * 60 * 60 * 1000
      ).toISOString()
    });

  } catch (error) {
    res.status(500).json({ 
      error: error.message 
    });
  }
});

const CacheIndicator = ({ 
  cachedAt, 
  expiresAt 
}) => {
  const hoursAgo = Math.floor(
    (Date.now() - new Date(cachedAt)) 
    / 3600000
  );
  const hoursLeft = Math.max(
    0,
    Math.ceil(
      (new Date(expiresAt) - Date.now()) 
      / 3600000
    )
  );

  const freshness = hoursAgo < 12 
    ? 'Fresh data' 
    : hoursAgo < 48 
    ? 'Recent data' 
    : 'Aging data';

  const color = hoursAgo < 12 
    ? '#10b981'
    : hoursAgo < 48 
    ? '#3b82f6' 
    : '#f59e0b';

  return (
    <div style={{ 
      fontSize: 11, 
      color, 
      marginTop: 8,
      display: 'flex',
      alignItems: 'center',
      gap: 8
    }}>
      <span>● {freshness}</span>
      <span style={{ color: '#9ca3af' }}>
        · Updated {hoursAgo}h ago
        · Refreshes in {hoursLeft}h
      </span>
    </div>
  );
};

async (req, res) => {
  const { domain, endpoint } = req.body;

  // Verify user owns this domain
  const { data: domainRecord } = await supabase
    .from('user_domains')
    .select('domain')
    .eq('user_id', req.user.id)
    .eq('domain', domain)
    .single();

  if (!domainRecord) {
    return res.status(403).json({
      error: 'Not authorized'
    });
  }

  await clearCache(domain, endpoint);
  res.json({ success: true });
});
