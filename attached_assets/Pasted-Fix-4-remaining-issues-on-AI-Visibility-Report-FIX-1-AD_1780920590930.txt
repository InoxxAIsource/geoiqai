const extractBrandName = (domain) => {
  const clean = domain
    .replace('www.','')
    .replace('https://','')
    .toLowerCase();
  
  const name = clean.split('.')[0];
  
  // Common prefixes to strip
  const prefixes = [
    'try','get','use','app','my',
    'go','join','meet','hello','hey'
  ];
  
  for (const prefix of prefixes) {
    if (name.startsWith(prefix) && 
        name.length > prefix.length + 2) {
      return name.slice(prefix.length);
    }
  }
  
  return name;
};

// getpostman → postman
