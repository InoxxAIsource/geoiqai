const loadUserContext = async (
  userId: string,
  domain: string
): Promise<UserContext> => {
  
  const [
    aiPresence,
    sprintProgress,
    answerMonitoring,
    brandPerformance,
    competitorData
  ] = await Promise.allSettled([
    
    // AI Visibility Report data
    supabase
      .from('geoiq_cache')
      .select('response_json, cached_at')
      .eq('domain', domain)
      .eq('endpoint', 'ai_presence')
      .order('cached_at', { ascending: false })
      .limit(1)
      .single(),
    
    // GEO Sprint progress
    supabase
      .from('sprint_progress')
      .select('step_id, completed, completed_at')
      .eq('user_id', userId)
      .eq('domain', domain),
    
    // Answer Monitoring prompts
    supabase
      .from('tracked_prompts')
      .select('prompt, mentioned, position, sentiment, context')
      .eq('user_id', userId)
      .eq('domain', domain)
      .limit(20),
    
    // Brand performance / signal tracker
    supabase
      .from('brand_performance')
      .select('*')
      .eq('domain', domain)
      .order('created_at', { ascending: false })
      .limit(1)
      .single(),
    
    // Latest competitor scores
    supabase
      .from('geoiq_cache')
      .select('response_json')
      .eq('domain', domain)
      .eq('endpoint', 'brand_benchmarks')
      .order('cached_at', { ascending: false })
      .limit(1)
      .single()
  ]);
  
  const presence = aiPresence.status === 'fulfilled'
    ? aiPresence.value.data?.response_json : null;
  
  const sprint = sprintProgress.status === 'fulfilled'
    ? sprintProgress.value.data : [];
  
  const prompts = answerMonitoring.status === 'fulfilled'
    ? answerMonitoring.value.data : [];
  
  const brand = brandPerformance.status === 'fulfilled'
    ? brandPerformance.value.data : null;
  
  const competitors = competitorData.status === 'fulfilled'
    ? competitorData.value.data?.response_json : null;
  
  // Calculate sprint stats
  const completedSteps = sprint
    .filter(s => s.completed).length;
  const totalSteps = 33;
  
  // Find next sprint step
  const SPRINT_STEPS = [...]; // import from geo-sprint
  const completedIds = new Set(
    sprint.filter(s => s.completed).map(s => s.step_id)
  );
  const nextStep = SPRINT_STEPS.find(
    s => !completedIds.has(s.id)
  );
  
  // Build context string for Claude
  return {
    domain,
    score: presence?.score || 0,
    scores: {
      chatgpt: presence?.systems?.chatgpt?.score || 0,
      gemini: presence?.systems?.gemini?.score || 0,
      perplexity: presence?.systems?.perplexity?.score || 0,
      claude: presence?.systems?.claude?.score || 0,
      grok: presence?.systems?.grok?.score || 0,
      google_ai: presence?.aio_score || 0
    },
    issues: presence?.issues || [],
    site_health: presence?.site_health_score || 0,
    ai_readiness: presence?.ai_readiness_score || 0,
    sprint: {
      completed: completedSteps,
      total: totalSteps,
      next_step: nextStep,
      phase: Math.floor(completedSteps / 5) + 1
    },
    prompts: {
      tracked: prompts?.length || 0,
      mentioned: prompts?.filter(p => p.mentioned).length || 0,
      best_prompt: prompts?.find(p => p.position === 1),
      negative_prompts: prompts?.filter(
        p => p.sentiment === 'negative'
      ) || []
    },
    competitors: competitors?.competitors || [],
    last_scan: presence?.cached_at || null
  };
};

const generateDailyBriefing = (ctx: UserContext) => {
  const parts = [];
  
  // Score status
  parts.push(`Your AI visibility score is 
    **${ctx.score}/100**`);
  
  // Best platform
  const best = Object.entries(ctx.scores)
    .sort((a,b) => b[1] - a[1])[0];
  parts.push(`Strongest platform: ${best[0]} 
    at ${best[1]}/100`);
  
  // Weakest platform
  const worst = Object.entries(ctx.scores)
    .filter(([,v]) => v > 0)
    .sort((a,b) => a[1] - b[1])[0];
  if (worst) {
    parts.push(`Biggest gap: ${worst[0]} 
      at ${worst[1]}/100`);
  }
  
  // Sprint recommendation
  if (ctx.sprint.next_step) {
    parts.push(`Recommended today: 
      ${ctx.sprint.next_step.title} 
      (+${ctx.sprint.next_step.score_impact} pts)`);
  }
  
  // Negative sentiment alert
  if (ctx.prompts.negative_prompts.length > 0) {
    parts.push(`⚠️ ${ctx.prompts.negative_prompts.length} 
      prompts showing negative AI sentiment`);
  }
  
  return parts.join('\n\n');
};

const predictScore = (
  currentScore: number,
  pendingSteps: SprintStep[]
) => {
  const topSteps = pendingSteps
    .sort((a,b) => b.score_impact - a.score_impact)
    .slice(0, 3);
  
  const totalGain = topSteps
    .reduce((sum, s) => sum + s.score_impact, 0);
  
  return {
    current: currentScore,
    predicted: Math.min(currentScore + totalGain, 99),
    steps: topSteps,
    time: topSteps.reduce(
      (sum, s) => sum + parseInt(s.time), 0
    )
  };
};

  // Add generated content as a special message

    contentType: type,
    label: `Generated ${type}`,
    copyButton: true,
    markDoneButton: true,
    sprintStepId: getSprintStepId(type)
  });
};
